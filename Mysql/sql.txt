create database gameAgent;

use gameAgent;

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT comment '用户Id',  -- 主键+自增
    account VARCHAR(50) NOT NULL comment '账号',          -- 非空
    password varchar(200) NOT NULL comment '密码',
    nickname varchar(20) default '匿名用户' comment '用户昵称',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP comment '创建时间'
) comment '用户表' AUTO_INCREMENT = 10000;

alter table users modify column account varchar(50) NOT NULL  unique  comment '账号';

create TABLE projects(
    id INT PRIMARY KEY AUTO_INCREMENT comment '项目id',
    account varchar(50) NOT NULL  comment '用户账号',
    dir_path varchar(200) NOT NULL comment '项目路径(绝对)',
    title varchar(50) NOT NULL comment '项目标题',
    `desc` text NOT NULL comment '项目描述',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP comment '创建时间'
)comment '项目表' auto_increment = 100;

alter table projects add column link varchar(100) comment '线上地址';
alter table projects add column current_vision int comment '当前线上版本ID';
alter table projects add column cloudflare_id varchar(50) comment 'cloudflareId 删除线上版本';
alter table projects add column temp_version varchar(50) default '1.0.0' comment '项目模板版本号';

// 建立索引的两种方式 ===========================================================
写法一：ALTER TABLE（常用）
USE gameAgent;
-- users 表：账号唯一索引
ALTER TABLE users ADD UNIQUE INDEX uk_users_account (account);
-- projects 表：账号普通索引
ALTER TABLE projects ADD INDEX idx_projects_account (account);
写法二：CREATE INDEX
USE gameAgent;
CREATE UNIQUE INDEX uk_users_account ON users (account);
CREATE INDEX idx_projects_account ON projects (account);
执行前建议先检查

USE gameAgent;
-- 1. 看表是否已有索引，避免重复创建
SHOW INDEX FROM users;
SHOW INDEX FROM projects;
-- 2. 创建唯一索引前，确认 account 没有重复数据
SELECT account, COUNT(*) AS cnt FROM users GROUP BY account HAVING cnt > 1;

create table sessions(
    id INT primary key auto_increment comment '会话id',
    message_id INT comment 'ai返回完整内容对应id',
    project_id INT not null comment '关联项目id',
    title varchar(50) comment '会话标题',
    account varchar(50) NOT NULL comment '用户账号',
    role varchar(20) NOT NULL comment '角色类型(用户/ai/vision)',
    content text comment '当前会话内容',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP comment '创建时间'
)comment '会话表';

create table messages(
    id INT primary key auto_increment comment '消息id',
    content text comment '消息完整内容',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP comment '创建时间'
)comment '消息表';

create table log(
    id INT primary key auto_increment comment '日志id',
    content text comment '日志内容',
    project_id INT not null comment '关联项目id',
    account varchar(50) not null comment '用户账号',
    created_at timestamp default current_timestamp comment '创建时间'
)comment '日志表' auto_increment = 1000;

-- 日志表增加会话标题字段，用于按会话过滤日志
ALTER TABLE log
  ADD COLUMN title VARCHAR(50) NULL COMMENT '会话标题' AFTER project_id;
CREATE INDEX idx_log_project_title ON log (project_id, account, title);

create table snapshots(
    id INT primary key auto_increment comment '快照id',
    project_id int not null comment '关联项目id',
    account varchar(50) not null comment '用户账号',
    version varchar(50) not null comment '版本号',
    `desc` varchar(100) comment '版本描述',
    file_path varchar(200) not null comment '文件绝对路径',
    file_content text comment '文件内容',
    bytes varchar(100) not null comment '文件大小',
    length int comment '文件行数',
    created_at timestamp default current_timestamp comment '创建时间'
)comment '快照表' auto_increment = 1000;

CREATE INDEX idx_snapshots_project_version ON snapshots (project_id, version);
alter table snapshots add column type int default 0 comment '快照类型(0：用户快照,1：版本升级前快照)';
alter table snapshots add column temp_version varchar(20) default '1.0.0' comment '快照模板版本';

// 双 Token 认证：Refresh Token 表 ===========================================================
USE gameAgent;

create table refresh_tokens(
    id INT primary key auto_increment comment '主键',
    user_id INT not null comment '用户ID',
    token_hash VARCHAR(64) not null comment 'refresh token 的 SHA256 哈希',
    expires_at TIMESTAMP not null comment '过期时间',
    revoked TINYINT default 0 comment '是否已吊销(0否1是)',
    created_at TIMESTAMP default CURRENT_TIMESTAMP comment '创建时间'
)comment '刷新令牌表';

CREATE INDEX idx_refresh_tokens_user_id ON refresh_tokens (user_id);
CREATE INDEX idx_refresh_tokens_hash ON refresh_tokens (token_hash);
